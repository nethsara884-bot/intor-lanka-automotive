        // SEARCH FUNCTIONALITY IN CATALOGUE
        function handleCatalogSearch() {
            const query = document.getElementById('catalogSearchInput').value.toLowerCase().trim();
            if(!query) {
                renderProductsGrid();
                return;
            }

            const filtered = productsData.filter(p => 
                p.title.toLowerCase().includes(query) || 
                p.code.toLowerCase().includes(query) || 
                p.category.toLowerCase().includes(query) ||
                p.brand.toLowerCase().includes(query)
            );

            renderProductsGrid(filtered);
        }

        // PRODUCT DETAIL MODAL & TWO-PHOTO SWITCHER
        function openProductDetailModal(id) {
            const product = productsData.find(p => p.id === id);
            if(!product) return;

            activeDetailProduct = product;
            selectedDetailPhotoIndex = 1;

            document.getElementById('detailTitle').textContent = product.title;
            document.getElementById('detailCode').textContent = product.code;
            document.getElementById('detailCategory').textContent = product.category;
            document.getElementById('detailBrand').textContent = product.brand;
            document.getElementById('detailPrice').textContent = product.price;
            document.getElementById('detailDescription').textContent = product.description;

            // Stock badge styling
            const stockBadge = document.getElementById('detailStockBadge');
            stockBadge.textContent = product.stock;
            if(product.stock === 'In Stock') {
                stockBadge.className = "absolute top-3 left-3 text-[10px] font-black uppercase px-2 py-1 rounded shadow bg-emerald-500 text-white";
            } else {
                stockBadge.className = "absolute top-3 left-3 text-[10px] font-black uppercase px-2 py-1 rounded shadow bg-amber-500 text-white";
            }

            // Set Photo Thumbnails
            document.getElementById('detailThumb1').src = product.img1;
            document.getElementById('detailThumb2').src = product.img2;
            switchDetailPhoto(1);

            // Set WhatsApp direct action link
            const waMsg = encodeURIComponent(`Hello INTOR LANKA, I wish to enquire about ordering ${product.title} (Code: ${product.code}).`);
            document.getElementById('detailWhatsAppLink').href = `https://wa.me/94778800709?text=${waMsg}`;

            document.getElementById('productDetailModal').classList.remove('hidden');
        }

        function switchDetailPhoto(index) {
            if(!activeDetailProduct) return;
            selectedDetailPhotoIndex = index;
            const mainImg = document.getElementById('detailMainImage');
            const btn1 = document.getElementById('photoBtn1');
            const btn2 = document.getElementById('photoBtn2');

            if(index === 1) {
                mainImg.src = activeDetailProduct.img1;
                btn1.className = "w-16 h-16 rounded-lg border-2 border-industrial-orange overflow-hidden bg-white p-1 focus:outline-none shadow-md";
                btn2.className = "w-16 h-16 rounded-lg border-2 border-transparent hover:border-slate-300 overflow-hidden bg-white p-1 focus:outline-none";
            } else {
                mainImg.src = activeDetailProduct.img2;
                btn2.className = "w-16 h-16 rounded-lg border-2 border-industrial-orange overflow-hidden bg-white p-1 focus:outline-none shadow-md";
                btn1.className = "w-16 h-16 rounded-lg border-2 border-transparent hover:border-slate-300 overflow-hidden bg-white p-1 focus:outline-none";
            }
        }

            document.getElementById('searchModal').classList.remove('hidden');
            document.getElementById('modalSearchInput').focus();
        }

        function handleModalSearch() {
            const query = document.getElementById('modalSearchInput').value.toLowerCase().trim();
            const resultsContainer = document.getElementById('modalSearchResults');

            if(!query) {
                resultsContainer.innerHTML = '<p class="text-slate-400 py-4 text-center">Type above to search automotive & machinery parts...</p>';
                return;
            }

            const matches = productsData.filter(p => 
                p.title.toLowerCase().includes(query) || 
                p.code.toLowerCase().includes(query) ||
                p.brand.toLowerCase().includes(query)
            );

            if(matches.length === 0) {
                resultsContainer.innerHTML = '<p class="text-slate-400 py-4 text-center">No parts match your search query.</p>';
                return;
            }

            resultsContainer.innerHTML = matches.map(m => `
                <div onclick="closeModal('searchModal'); openProductDetailModal('${m.id}')" class="py-2.5 px-2 hover:bg-slate-100 cursor-pointer flex items-center justify-between transition">
                    <div>
                        <span class="font-extrabold text-navy-900 block">${m.title}</span>
                        <span class="text-[10px] text-slate-500 font-mono">Code: ${m.code} | Category: ${m.category}</span>
                    </div>
                    <span class="text-xs font-bold text-industrial-orange shrink-0 ml-2">${m.price}</span>
                </div>
            `).join('');
        }

        // ADMIN VIEW SWITCHING & AUTH
        function toggleAdminView() {
            const adminView = document.getElementById('adminDashboardView');
            const publicView = document.getElementById('publicSiteView');

            if(adminView.classList.contains('hidden')) {
                // Simple PIN protection for store administrator
                const pin = prompt("Enter INTOR LANKA Admin PIN (Default: 1234):");
                if(pin === "1234") {
                    publicView.classList.add('hidden');
                    adminView.classList.remove('hidden');
                    renderAdminTable();
                } else if(pin !== null) {
                    alert("Incorrect Admin PIN.");
                }
            } else {
                adminView.classList.add('hidden');
                publicView.classList.remove('hidden');
            }
        }

        function renderAdminTable() {
            const tbody = document.getElementById('adminProductsTableBody');
            tbody.innerHTML = productsData.map(item => `
                <tr class="hover:bg-slate-800/50 transition">
                    <td class="p-4">
                        <img src="${item.img1}" class="w-10 h-10 object-cover rounded bg-slate-950 border border-slate-700" onerror="this.src='https://placehold.co/100/0a192f/ffffff?text=Part'">
                    </td>
                    <td class="p-4 font-extrabold text-white">${item.title}</td>
                    <td class="p-4 font-mono text-slate-400">${item.code}</td>
                    <td class="p-4 text-slate-300">${item.category}</td>
                    <td class="p-4 text-industrial-orange font-bold">${item.brand}</td>
                    <td class="p-4 font-bold text-white">${item.price}</td>
                    <td class="p-4">
                        <span class="px-2 py-0.5 rounded text-[10px] font-bold ${item.stock === 'In Stock' ? 'bg-emerald-950 text-emerald-400 border border-emerald-800' : 'bg-amber-950 text-amber-400 border border-amber-800'}">
                            ${item.stock}
                        </span>
                    </td>
                    <td class="p-4 text-right space-x-2">
                        <button onclick="editProductAdmin('${item.id}')" class="text-blue-400 hover:text-blue-300 font-bold">Edit</button>
                        <button onclick="deleteProductAdmin('${item.id}')" class="text-rose-400 hover:text-rose-300 font-bold">Delete</button>
                    </td>
                </tr>
            `).join('');

            updateStats();
        }

        function updateStats() {
            document.getElementById('statTotalItems').textContent = productsData.length;
            document.getElementById('statInStock').textContent = productsData.filter(p => p.stock === 'In Stock').length;
            document.getElementById('statLowStock').textContent = productsData.filter(p => p.stock !== 'In Stock').length;
        }

        function filterAdminTable() {
            const query = document.getElementById('adminTableSearch').value.toLowerCase();
            const rows = document.querySelectorAll('#adminProductsTableBody tr');
            rows.forEach(row => {
                const text = row.textContent.toLowerCase();
                row.style.display = text.includes(query) ? '' : 'none';
            });
        }

        function openAddProductModal() {
            document.getElementById('adminModalTitle').textContent = "ADD NEW SPARE PART";
            document.getElementById('adminProductForm').reset();
            document.getElementById('editProductId').value = '';
            document.getElementById('adminProductModal').classList.remove('hidden');
        }

        function editProductAdmin(id) {
            const item = productsData.find(p => p.id === id);
            if(!item) return;

            document.getElementById('adminModalTitle').textContent = "EDIT SPARE PART";
            document.getElementById('editProductId').value = item.id;
            document.getElementById('formTitle').value = item.title;
            document.getElementById('formCode').value = item.code;
            document.getElementById('formCategory').value = item.category;
            document.getElementById('formBrand').value = item.brand;
            document.getElementById('formPrice').value = item.price;
            document.getElementById('formStock').value = item.stock;
            document.getElementById('formImg1').value = item.img1;
            document.getElementById('formImg2').value = item.img2;
            document.getElementById('formDescription').value = item.description;

            document.getElementById('adminProductModal').classList.remove('hidden');
        }

        function saveProductFromAdmin(event) {
            event.preventDefault();
            const editId = document.getElementById('editProductId').value;

            const productObj = {
                id: editId ? editId : 'p_' + Date.now(),
                title: document.getElementById('formTitle').value,
                code: document.getElementById('formCode').value,
                category: document.getElementById('formCategory').value,
                brand: document.getElementById('formBrand').value,
                price: document.getElementById('formPrice').value,
                stock: document.getElementById('formStock').value,
                img1: document.getElementById('formImg1').value,
                img2: document.getElementById('formImg2').value,
                description: document.getElementById('formDescription').value
            };

            if(editId) {
                const idx = productsData.findIndex(p => p.id === editId);
                if(idx !== -1) productsData[idx] = productObj;
            } else {
                productsData.unshift(productObj);
            }

            saveProductsToStorage();
            closeModal('adminProductModal');
            renderAdminTable();
            renderCategoriesGrid();
            renderProductsGrid();
        }

        function deleteProductAdmin(id) {
            if(confirm("Are you sure you want to delete this spare part from the inventory?")) {
                productsData = productsData.filter(p => p.id !== id);
                saveProductsToStorage();
                renderAdminTable();
                renderCategoriesGrid();
                renderProductsGrid();
            }
        }
